def substitutions(text):

        O -> O / 0

    Ejemplo:

        Shadow

    genera:

        Shadow
        Shad0w
        Sh@dow
        Sh@d0w
    """

    options = []

    for char in text:

        if char == "a":
            options.append(["a", "@"])

        elif char == "A":
            options.append(["A", "@"])

        elif char == "o":
            options.append(["o", "0"])

        elif char == "O":
            options.append(["O", "0"])

            options.append([char])

    return {
        "".join(combination)
        for combination in product(*options)
